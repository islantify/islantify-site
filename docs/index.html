    document.addEventListener("DOMContentLoaded", () => { // 1. Matcha elementen direkt utifrån klasserna i din befintliga HTML const inputTimlon = document.querySelector(".demo-input"); const btnStart = document.querySelector(".demo-btn-start"); const btnPause = document.querySelector(".demo-btn-pause"); const btnReset = document.querySelector(".demo-btn-reset"); // Hittar inkomst-displayerna baserat på textinnehållet i din HTML let displayLiveInkomst = null; let displayKrSekund = null; document.querySelectorAll("div, p, span, strong").forEach(el => { if (el.textContent.includes("0,00 kr")) { displayLiveInkomst = el; } if (el.textContent.includes("kr/sek")) { displayKrSekund = el; } }); let intervalId = null; let upplupenInkomst = 0.0; // 2. Beräkna och uppdatera kronor per sekund (+ 0.0694 kr/sek vid 250 kr) function uppdateraSekundlon() { const timlon = parseFloat(inputTimlon?.value) || 0; const krPerSekund = timlon / 3600; // 60 minuter * 60 sekunder if (displayKrSekund) { displayKrSekund.innerHTML = `+ ${krPerSekund.toFixed(4)} kr/sek`; } return krPerSekund; } // Lyssna på ändringar i input-fältet direkt if (inputTimlon) { inputTimlon.addEventListener("input", uppdateraSekundlon); uppdateraSekundlon(); // Sätt startvärde direkt vid laddning } // 3. STARTA-knapp (Räknar upp mjukt var 10:e del av en sekund) if (btnStart) { btnStart.addEventListener("click", () => { if (intervalId) return; // Redan igång if (inputTimlon) inputTimlon.disabled = true; // Lås input under körning const uppdateringsFrekvens = 0.1; // 100ms för ett mjukt flyt intervalId = setInterval(() => { const nuvarandeKrPerSekund = uppdateraSekundlon(); upplupenInkomst += nuvarandeKrPerSekund * uppdateringsFrekvens; if (displayLiveInkomst) { displayLiveInkomst.innerHTML = `${upplupenInkomst.toLocaleString("sv-SE", { minimumFractionDigits: 2, maximumFractionDigits: 2 })} kr`; } }, uppdateringsFrekvens * 1000); }); } // 4. PAUSA-knapp if (btnPause) { btnPause.addEventListener("click", () => { clearInterval(intervalId); intervalId = null; if (inputTimlon) inputTimlon.disabled = false; }); } // 5. NOLLSTÄLLA (Om du har eller lägger till en återställningsknapp) if (btnReset) { btnReset.addEventListener("click", () => { clearInterval(intervalId); intervalId = null; upplupenInkomst = 0.0; if (displayLiveInkomst) displayLiveInkomst.innerHTML = "0,00 kr"; if (inputTimlon) inputTimlon.disabled = false; uppdateraSekundlon(); }); } });
